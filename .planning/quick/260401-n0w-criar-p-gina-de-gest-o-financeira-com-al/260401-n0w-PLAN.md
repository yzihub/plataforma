3. Implement `fetchFinanceRecords()` async function following the leads pattern exactly:
   ```
   const supabase = await createClient();
   const { data: { user } } = await supabase.auth.getUser();
   if (!user) return MOCK_FINANCE;

   const { data: profile } = await supabase
     .from("profiles")
     .select("tenant_id")
     .eq("id", user.id)
     .single();

   if (!profile?.tenant_id) return MOCK_FINANCE;

   const { data, error } = await supabase
     .from("finance")
     .select("id, tenant_id, final_amount, financial_alert, priority_flag")
     .eq("tenant_id", profile.tenant_id)
     .order("created_at", { ascending: false });

   if (error || !data) return MOCK_FINANCE;
   return data as FinanceRecord[];
   ```
   Wrap entire function in try/catch returning MOCK_FINANCE on error.
